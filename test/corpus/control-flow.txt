fn f(xs: List(Int)) -> Int {
    let total = for (sum = 0) x in xs.iter() { sum + x }
    let found = for (seen = 0) x in xs.iter() {
        if x < 0 { break seen }
        if x == 0 { continue }
        seen + 1
    }
    for x in xs.iter() { io::print(x.to_str()) }
    total + found
}
