fn get(db: Db, id: Int, ids: List(Int)) -> Result(User, sql::Error) {
    db.one(sql(User) { SELECT * FROM users WHERE id = {id} })
    db.all(sql(User) { SELECT id, email FROM users WHERE id = ANY({ids}) ORDER BY id LIMIT {limit} OFFSET {offset} })
}

=========================================================================
a query with no row type, answered as a count, and one decoding a column
=========================================================================

fn touch(db: Db, now: Timestamp, id: Int, new: NewUser) -> Result(Int, sql::Error) {
    try db.exec(sql { UPDATE threads SET last_activity = {now} WHERE id = {id} })
    db.one(sql(Int) { INSERT INTO users {new} RETURNING id })
}

fn update(db: Db, user: User) -> Result(Int, sql::Error) {
    db.exec(
        sql {
            UPDATE users
            SET {user}
            WHERE id = {user.id}
              AND email LIKE {pattern}
        }
        .label("users.update"),
    )
}
