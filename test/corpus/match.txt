fn f() string {
	match m {
		On { value } => value
		Move { x: a, y: b } => a
		Config { x, .. } => x
		Pair(a, b) => a
		None => "x"
		_ => "y"
	}
}

fn f() {
	let x = match v {
		1 => "one",
		"x" => "ex",
		true => "yes",
		_ => { panic() }
	}
}
