    fn begin(self) -> Active {
        Active { n: self.n }

        Idle { n: self.n }

==================================================================================
the state as a type parameter, with a blanket impl for what every state shares
==================================================================================

struct Idle {}
struct Active {}
linear struct Conn<S> { n: Int }

impl<S> Conn<S> {
    fn tally(self) -> Int {
        self.n
    }
}

impl Conn<Idle> {
    fn begin(self) -> Conn<Active> {
        Conn { n: self.n }
    }
}
